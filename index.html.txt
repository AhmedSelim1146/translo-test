// 6. TIMER LOGIC
function startTimer() {
clearInterval(timerInterval);
timerInterval = setInterval(() => {
if (timerSeconds <= 0) {
clearInterval(timerInterval);
showToast("انتهى الوقت المخصص للاختبار! جاري عرض النتيجة...");
finalizeAndGradeTest();
return;
}
timerSeconds--;
const mins = Math.floor(timerSeconds / 60);
const secs = timerSeconds % 60;
const formatted = `${String(mins).padStart(2, "0")}:${String(secs).padStart(2, "0")}`;
const display = document.getElementById("timer-display");
display.textContent = formatted;
if (timerSeconds < 180) {
display.classList.add("text-rose-400");
}
}, 1000);
}

// 7. SPEECH SYNTHESIS (AUDIO PLAYER FOR LISTENING)
function playAudio(text) {
if (!("speechSynthesis" in window)) {
showToast("متصفحك لا يدعم التشغيل الصوتي التلقائي. يمكنك استخدام زر إظهار النص الصوتي.");
return;
}
window.speechSynthesis.cancel();
const utterance = new SpeechSynthesisUtterance(text);
utterance.lang = "en-US";
utterance.rate = speechRate;

const voices = window.speechSynthesis.getVoices();
const preferredVoice = voices.find(v => v.lang.includes("en-US") && (v.name.includes("Google") || v.name.includes("Natural") || v.name.includes("Samantha"))) || voices.find(v => v.lang.startsWith("en"));
if (preferredVoice) utterance.voice = preferredVoice;

utterance.onstart = () => {
isSpeaking = true;
updateAudioUIState();
};
utterance.onend = () => {
isSpeaking = false;
updateAudioUIState();
};
utterance.onerror = () => {
isSpeaking = false;
updateAudioUIState();
};

window.speechSynthesis.speak(utterance);
}

function stopAudio() {
if ("speechSynthesis" in window) {
window.speechSynthesis.cancel();
}
isSpeaking = false;
updateAudioUIState();
}

function toggleSpeechRate() {
speechRate = speechRate === 1.0 ? 0.85 : 1.0;
const btn = document.getElementById("speed-toggle-btn");
if (btn) btn.textContent = speechRate === 1.0 ? "1.0x عادي" : "0.85x بطيء";
if (isSpeaking) {
const q = questions[currentIndex];
if (q && q.trackId) playAudio(listeningTracks[q.trackId].script);
}
}

function toggleTranscriptVisibility() {
showTranscript = !showTranscript;
const box = document.getElementById("transcript-box");
const btn = document.getElementById("transcript-toggle-btn");
if (box && btn) {
box.classList.toggle("hidden", !showTranscript);
btn.textContent = showTranscript ? "إخفاء النص الصوتي" : "إظهار النص الصوتي (للمساعدة)";
}
}

function updateAudioUIState() {
const wave = document.getElementById("audio-waveform");
const statusText = document.getElementById("audio-status-text");

statusText.textContent = isSpeaking ? "جاري تشغيل المقطع الصوتي..." : "اضغط تشغيل للاستماع إلى المحادثة";
}
}

btnClass = "bg-teal-600 text-white border-teal-600";
}

const btn = document.createElement("button");
btn.type = "button";
btn.textContent = q.id;
btn.className = `en-text py-2 rounded-xl text-xs font-bold transition cursor-pointer ${btnClass}`;
btn.onclick = () => jumpToQuestion(idx);

grids[q.section].appendChild(btn);
});
}

// 14. ANSWER REVIEW RENDERER
function toggleAnswerReview() {
const reviewSection = document.getElementById("answer-review-section");
const btnText = document.getElementById("toggle-review-text");
const isHidden = reviewSection.classList.contains("hidden");

if (isHidden) {
renderReviewList("all");
reviewSection.classList.remove("hidden");
btnText.textContent = "إخفاء مراجعة الإجابات";
reviewSection.scrollIntoView({ behavior: "smooth" });
} else {
reviewSection.classList.add("hidden");
btnText.textContent = "عرض مراجعة الإجابات والتصحيح";
}
}

function renderReviewList(filterMode) {
const container = document.getElementById("review-items-list");
const letters = ["A", "B", "C", "D"];
const filtered = questions.map((q, idx) => ({ q, idx })).filter(({ q, idx }) => {
if (filterMode === "wrong") return userAnswers[idx] !== q.correct;
return true;
});

if (filtered.length === 0) {
container.innerHTML = `<div class="p-6 text-center text-sm font-bold text-teal-700 bg-teal-50 rounded-2xl">ممتاز! لا توجد أي أخطاء لعرضها.</div>`;
return;
}

container.innerHTML = filtered.map(({ q, idx }) => {
const userAns = userAnswers[idx];
const isCorrect = userAns === q.correct;
const userText = userAns !== null ? `${letters[userAns]}. ${q.options[userAns]}` : "لم تتم الإجابة (Unanswered)";
const correctText = `${letters[q.correct]}. ${q.options[q.correct]}`;

return `
<div class="p-4 sm:p-5 rounded-2xl border ${isCorrect ? "border-emerald-200 bg-emerald-50/30" : "border-rose-200 bg-rose-50/30"}">
<div class="flex flex-wrap items-center justify-between gap-2 mb-2">
<div class="flex items-center gap-2">
<span class="text-xs font-extrabold px-2.5 py-0.5 rounded-lg bg-slate-900 text-white">سؤال ${q.id}</span>
<span class="en-text text-xs font-bold px-2 py-0.5 rounded bg-slate-200 text-slate-700">${q.level}</span>
</div>
<span class="text-xs font-bold px-2.5 py-1 rounded-full ${isCorrect ? "bg-emerald-100 text-emerald-800" : "bg-rose-100 text-rose-800"}">
${isCorrect ? "إجابة صحيحة ✓" : "إجابة غير صحيحة ✕"}
</span>
</div>

<p class="en-text text-sm sm:text-base font-bold text-slate-900 mb-3">${q.prompt}</p>

<div class="grid grid-cols-1 sm:grid-cols-2 gap-2 text-xs sm:text-sm mb-3">
<div class="en-text p-2.5 rounded-xl bg-white border ${isCorrect ? "border-emerald-300 text-emerald-800" : "border-rose-300 text-rose-800"}">
<strong class="block text-[11px] uppercase tracking-wider opacity-70">Your Answer:</strong>
${userText}
</div>
<div class="en-text p-2.5 rounded-xl bg-white border border-emerald-300 text-emerald-900">
<strong class="block text-[11px] uppercase tracking-wider opacity-70">Correct Answer:</strong>
${correctText}
</div>
</div>

<p class="en-text text-xs text-slate-600 bg-white/80 p-2.5 rounded-xl border border-slate-200/80">
<strong>Explanation:</strong> ${q.explanation}
</p>
</div>
`;
}).join("");
}

function copyResultSummary() {
const score = document.getElementById("res-total-score").textContent;
const pct = document.getElementById("res-percentage").textContent;
const code = document.getElementById("res-cefr-code").textContent;
const title = document.getElementById("res-cefr-title").textContent;

const summary = `نتيجة اختبار تحديد مستوى اللغة الإنجليزية - Translo\n${nameLine}الدرجة: ${score}/50 (${pct})\nالمستوى المعتمد: ${code} (${title})`;

const tempInput = document.createElement("textarea");
tempInput.value = summary;
document.body.appendChild(tempInput);
tempInput.select();
try {
document.execCommand("copy");
showToast("تم نسخ ملخص النتيجة إلى الحافظة بنجاح!");
} catch (e) {
showToast("تعذر النسخ التلقائي، يمكنك طباعة التقرير.");
}
document.body.removeChild(tempInput);
}
